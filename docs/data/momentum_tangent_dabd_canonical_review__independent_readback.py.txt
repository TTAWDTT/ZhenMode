def time_coefficients(times,dtype):

def weighted_rms(error,weight,times,dtype=np.float64):
 e=np.asarray(error,dtype=dtype);w=np.broadcast_to(np.asarray(weight,dtype=dtype),e.shape)
 if e.shape[0]!=len(times) or not np.all(np.isfinite(e)) or not np.all(np.isfinite(w)) or not np.all(w>0):raise ValueError('invalid error/weight/time interface')
 ef=e.reshape(e.shape[0],-1);wf=w.reshape(w.shape[0],-1)
 # Separate normalized spatial weights at each output time, then time quadrature.
 spatial=wf/np.sum(wf,axis=1,dtype=dtype)[:,None]
 frame=np.sum(spatial*ef**2,axis=1,dtype=dtype)
 return np.sqrt(np.dot(time_coefficients(times,dtype),frame))
