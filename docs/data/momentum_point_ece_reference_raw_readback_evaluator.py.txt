def integrated_rms(field,weights,times):
    # Explicit per-time weighted spatial sum, then trapezoidal time quadrature.
    space=np.array([np.average(np.ravel(values*values),weights=np.ravel(weight)) for values,weight in zip(field,np.broadcast_to(weights,field.shape))])
    return float(np.sqrt(np.trapezoid(space,times)/(times[-1]-times[0])))
