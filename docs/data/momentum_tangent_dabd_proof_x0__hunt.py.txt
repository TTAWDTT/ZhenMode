# One-shot spec-only adversarial finite screen; no general-n proof claimed.
import sympy as s
import itertools, time, json
from pathlib import Path
started=time.monotonic(); rows=[]
def check(cond, label):
    if not cond: raise AssertionError('FIRE_OR_GATE_FAILURE '+label)
def zero(M): return all(s.simplify(e)==0 for e in M)
for n in (4,5,8):
    tick=time.monotonic()
    R=s.zeros(n-2,n)
    for j in range(n-2):
        for q,c in enumerate((1,-2,1)): R[j,j+q]=c
    L=s.diag(*([-3]+[-2]*(n-2)+[-3]))
    for j in range(n-1): L[j,j+1]=L[j+1,j]=1
    I=s.eye(n); Ft=I-R.T*R/16; Fn=I-L*L/16; B=(I+L/4)**2
    one=s.ones(n,1); affine=s.Matrix([s.Rational(2*j+1,2*n) for j in range(n)])
    check(Ft==Ft.T and Fn==Fn.T,'symmetry n='+str(n))
    check(Ft*one==one and Ft*affine==affine and one.T*Ft==one.T,'preservation n='+str(n))
    check(R.rank()==n-2 and (I-Ft).rank()==n-2,'nullspace/eigenvalue-one n='+str(n))
    mins={}; counts={}
    for label,A in [('Ft',Ft),('I-Ft',I-Ft),('Fn',Fn),('I-Fn',I-Fn),('I-Ft2',I-Ft**2),('I-Fn2',I-Fn**2)]:
        values=[]
        for size in range(1,n+1):
            for inds in itertools.combinations(range(n),size):
                d=A.extract(inds,inds).det(); check(d>=0,label+' negative minor n='+str(n)+' inds='+str(inds)+' det='+str(d)); values.append(d)
        mins[label]=str(min(values)); counts[label]=len(values)
    # Rebuild R transpose addition without reading matrix entries.
    for col in range(n):
        x=I[:,col]; dd=[x[j]-2*x[j+1]+x[j+2] for j in range(n-2)]; out=list(x)
        for j,d in enumerate(dd):
            for q,c in enumerate((1,-2,1)): out[j+q]-=c*d/16
        check(s.Matrix(out)==Ft*x,'direct second-difference route n='+str(n))
    # Exact trigonometric endpoint and k=n attacks.
    for k in range(1,n+1):
        v=s.Matrix([s.sin(s.pi*k*s.Rational(2*j+1,2*n)) for j in range(n)])
        lam=-4*s.sin(s.pi*k/(2*n))**2
        check(zero(L*v-lam*v),'L sine n='+str(n)+' k='+str(k))
        check(zero(Fn*v-(1-s.sin(s.pi*k/(2*n))**4)*v),'Fn sine n='+str(n)+' k='+str(k))
        check(zero(B*v-s.cos(s.pi*k/(2*n))**4*v),'B sine n='+str(n)+' k='+str(k))
    alt=s.Matrix([(-1)**j for j in range(n)]); check(Fn*alt==s.zeros(n,1),'nyquist n='+str(n))
    row={'n':n,'minimum_principal_minors':mins,'principal_minor_counts':counts,'sine_cases':n,'direct_basis_cases':n,'elapsed_seconds':time.monotonic()-tick,'status':'NO-FIRE finite screen'}
    rows.append(row); Path('hunt.results.json').write_text(json.dumps(rows,indent=2)); print(json.dumps(row),flush=True)
# Additional already pinned n4 counter-overclaims.
n=4; R=s.Matrix([[1,-2,1,0],[0,1,-2,1]]); Ft=s.eye(4)-R.T*R/16
L=s.Matrix([[-3,1,0,0],[1,-2,1,0],[0,1,-2,1],[0,0,1,-3]]); Fn=s.eye(4)-L*L/16
TV=lambda x:sum(abs(x[j+1]-x[j]) for j in range(len(x)-1))
e0=s.Matrix([1,0,0,0]); one=s.ones(4,1); e1=s.Matrix([0,1,0,0]); W=s.diag(1000,1,1,1)
check(TV(Ft*e0)>TV(e0),'TVD tangential witness')
check(TV(Fn*one)>TV(one),'TVD normal witness')
weighted=(e1.T*Ft.T*W*Ft*e1)[0]; check(weighted>1,'arbitrary weights witness')
print('OUT_OF_TARGET Ft e0='+str(list(Ft*e0))+' TV='+str(TV(Ft*e0))+' inputTV=1; Fn ones TV='+str(TV(Fn*one))+' inputTV=0; W=diag(1000,1,1,1) Ft e1 weightednorm2='+str(weighted)+' input=1',flush=True)
th=s.symbols('theta',real=True)
print('SYMBOL_CHECK normal_departure='+str(s.series(s.sin(th/2)**4,th,0,6))+' original_departure='+str(s.series(1-s.cos(th/2)**4,th,0,6)),flush=True)
print('X0_NO_FIRE finite cases only; general-n proof OPEN; total_seconds='+str(time.monotonic()-started),flush=True)
