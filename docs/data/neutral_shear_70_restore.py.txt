"""Restore exact archived bytes into a NEW directory; never execute native runs."""
import argparse
import hashlib
import json
from pathlib import Path, PurePosixPath
import tempfile
import zipfile

def sha_file(path):
    h = hashlib.sha256()
    with path.open('rb') as f:
        for block in iter(lambda: f.read(1048576), b''):
            h.update(block)
    return h.hexdigest()

def restore(output, manifest_path):
    m = json.loads(manifest_path.read_text(encoding='utf-8'))
    if output.exists():
        raise FileExistsError('Use a new output directory; existing files are never replaced.')
    with tempfile.TemporaryDirectory(prefix='neutral-shear-restore-') as temporary:
        archive = Path(temporary) / 'originals.zip'
        with archive.open('xb') as out:
            for part in m['parts']:
                name = part['file']
                assert Path(name).name == name
                p = manifest_path.parent / name
                assert p.stat().st_size == part['bytes'] and sha_file(p) == part['sha256'], name
                with p.open('rb') as f:
                    for block in iter(lambda: f.read(1048576), b''):
                        out.write(block)
        assert archive.stat().st_size == m['archive_bytes'] and sha_file(archive) == m['archive_sha256']
        expected = {r['member']: r for r in m['original_files']}
        assert len(expected) == len(m['original_files'])
        with zipfile.ZipFile(archive) as z:
            assert len(z.namelist()) == len(expected) and set(z.namelist()) == set(expected)
            # Validate every name and payload before writing restored members.
            for name, row in expected.items():
                pure = PurePosixPath(name)
                assert not pure.is_absolute() and '..' not in pure.parts and '\\' not in name and ':' not in name
                data = z.read(name)
                assert len(data) == row['bytes'] and hashlib.sha256(data).hexdigest() == row['sha256'], name
            output.mkdir(parents=True, exist_ok=False)
            for name, row in expected.items():
                dest = output.joinpath(*PurePosixPath(name).parts)
                dest.parent.mkdir(parents=True, exist_ok=True)
                with dest.open('xb') as f:
                    f.write(z.read(name))
                assert sha_file(dest) == row['sha256'], name
    print(f'Restored {len(expected)} exact original files. No scientific computation performed.')

if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument('--output', required=True, type=Path)
    parser.add_argument('--manifest', type=Path, default=Path(__file__).with_name('neutral_shear_70_manifest.json'))
    args = parser.parse_args()
    restore(args.output, args.manifest)
