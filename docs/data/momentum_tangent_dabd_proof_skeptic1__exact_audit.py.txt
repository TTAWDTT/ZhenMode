def encode(v):
    if isinstance(v,s.MatrixBase): return [[str(v[i,j]) for j in range(v.cols)] for i in range(v.rows)]
    if isinstance(v,(list,tuple)): return [encode(t) for t in v]
    if isinstance(v,dict): return {str(k):encode(t) for k,t in v.items()}
    return str(v)
def record(name,ok,values=None):
    item={'name':name,'pass':bool(ok),'values':encode(values)}
    checks.append(item)
    (out/'controls.json').write_text(json.dumps(checks,indent=2)+'\n')
    assert ok, item
def residual(v):
    if isinstance(v,s.MatrixBase): return v.applyfunc(lambda t:s.trigsimp(s.simplify(t)))
    return s.trigsimp(s.expand_trig(s.simplify(v)))
def equal(name,a,b):
    r=residual(a-b); zero=s.zeros(*r.shape) if isinstance(r,s.MatrixBase) else 0
    record(name,r==zero,{'actual':a,'expected':b,'residual':r})
def psd_bad(A):
    import itertools
    if A!=A.T: return {'reason':'nonsymmetric'}
    for k in range(1,A.rows+1):
        for subset in itertools.combinations(range(A.rows),k):
            val=s.factor(A.extract(subset,subset).det())
            if val<0:return {'principal_indices':subset,'determinant':val}
    return None
def contract_bad(A):return psd_bad(s.eye(A.rows)-A.T*A)
def tv(v):return sum(abs(v[j+1]-v[j]) for j in range(len(v)-1))
def weighted(v,W):return (v.T*W*v)[0]
