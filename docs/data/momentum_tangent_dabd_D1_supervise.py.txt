"""Single attempt, single CPU, bounded sequential D1 subprocess supervisor."""
import hashlib,json,os,pathlib,signal,subprocess,sys,time

ROOT=pathlib.Path(__file__).resolve().parent
NATIVE='/root/.venvs/zhenmode-managed-bench-20261005/bin/python'
SUITE=sys.executable


def rss_tree(pid):
    parents={};memory={}
    for proc in pathlib.Path('/proc').iterdir():
        if not proc.name.isdigit():continue
        try:
            status=(proc/'status').read_text();values={line.split(':',1)[0]:line.split(':',1)[1].strip() for line in status.splitlines() if ':' in line}
            parents[int(proc.name)]=int(values['PPid']);memory[int(proc.name)]=int(values.get('VmRSS','0 kB').split()[0])*1024
        except (OSError,KeyError,ValueError):pass
    descendants={pid}
    while True:
        new=descendants|{p for p,parent in parents.items() if parent in descendants}
        if new==descendants:break
        descendants=new
    return sum(memory.get(p,0) for p in descendants),sorted(descendants)


def stage(name,command,start):
    log=ROOT/(name+'.log');resources=ROOT/(name+'-resources.jsonl')
    with log.open('x') as stdout,resources.open('x') as samples:
        proc=subprocess.Popen(command,cwd=ROOT,stdout=stdout,stderr=subprocess.STDOUT,start_new_session=True)
        peak=0;reason=None
        while proc.poll() is None:
            memory,pids=rss_tree(proc.pid);peak=max(peak,memory)
            elapsed=time.monotonic()-start
            samples.write(json.dumps(dict(elapsed_s=elapsed,rss_bytes=memory,pids=pids))+'\n');samples.flush()
            if elapsed>180:reason='aggregate wall cap180s'
            if memory>4*1024**3:reason='sampled RSS cap4GiB'
            if name=='timing' and elapsed-stage_start>60 and 'first_helper' not in log.read_text():reason='no first helper within60s'
            if reason:
                os.killpg(proc.pid,signal.SIGKILL);break
            time.sleep(.2)
        rc=proc.wait()
    row=dict(stage=name,returncode=rc,peak_rss_bytes=peak,elapsed_total_s=time.monotonic()-start,stop_reason=reason,command=command)
    with (ROOT/'scoreboard.jsonl').open('a') as f:f.write(json.dumps(row)+'\n')
    print(json.dumps(row),flush=True)
    if rc or reason:raise RuntimeError('stage stopped: '+name)
    return row
