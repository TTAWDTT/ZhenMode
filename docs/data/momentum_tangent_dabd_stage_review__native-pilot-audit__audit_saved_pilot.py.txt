def require(condition, name):

def identical(a, b): return a.shape == b.shape and a.dtype == b.dtype and np.array_equal(a, b)
def arithmetic_equal(a, b, inventory=False): return abs(a-b) <= (2e-14 if inventory else 5e-15 + 5e-12*abs(b))
def original_times(times, contract): return identical(times, np.arange(0., contract['duration_s'] + 1., contract['output_s']))

def npz(path):
    with zipfile.ZipFile(path) as z:
        require(sum(x.file_size for x in z.infolist()) < 128*1024**2, 'bounded decoded NPZ')
    with np.load(path, allow_pickle=False) as z:

def weighted_rms(error, weights, times, duration):
    # Scalar compensated accumulation is separately expressed from the project
    # scorer and the earlier70 dot-product audit: each sample mean-square first,
    # then trapezoid contributions divided by the contract duration.
    mean_squares = []
    for j in range(len(times)):
        w = np.broadcast_to(weights[j], error[j].shape).reshape(-1)
        e = error[j].reshape(-1)
        denominator = math.fsum(float(v) for v in w)
        require(denominator > 0, 'positive spatial weight')
        numerator = math.fsum(float(ww)*float(ee)*float(ee) for ww, ee in zip(w, e))
        mean_squares.append(numerator / denominator)
    value = math.fsum((mean_squares[j-1]+mean_squares[j]) * float(times[j]-times[j-1]) / 2. for j in range(1,len(times)))
    return math.sqrt(value / duration)

def analytic_reference(c, a):
    q = c['internal_wave']['q']; m = q/c['H_m']; k = math.pi/c['Ly_m']; amplitude = c['amplitude_m']; gamma = c['internal_wave']['dT_dz_C_m']
    omega = math.sqrt(c['f']**2 + c['internal_wave']['N2_s_2']*(k/m)**2)
    z = np.linspace(0., -c['H_m'], c['nz']); phase = omega*a['time']
    ref = {'eta':amplitude*math.sin(q)*np.outer(np.cos(phase),np.cos(k*a['y_eta'])),
           'S':np.full_like(a['S'],c['S_psu'])}
    displacement = amplitude*np.cos(phase)[:,None,None]*np.cos(k*a['y_eta'])[None,:,None]*np.sin(m*(z+c['H_m']))[None,None,:]
    ref['T'] = (c['T_C']+gamma*z)[None,None,:] - gamma*displacement
    for field in ('u','v'):
        mode = np.sin(k*a['y_'+field])[:,None]*np.cos(m*(z+c['H_m']))[None,:]
        evolution = -c['f']*np.cos(phase) if field == 'u' else omega*np.sin(phase)
        ref[field] = amplitude*(m/k)*evolution[:,None,None]*mode[None,:,:]
    return ref, omega

def independently_measure(c, a):
    ref, omega = analytic_reference(c, a)
    weights_TS = a['h']*a['area'][None,:,None]
    weights_eta = np.broadcast_to(a['area'],a['eta'].shape)
    absolute = {'eta':weighted_rms(a['eta']-ref['eta'],weights_eta,a['time'],c['duration_s'])}
    for field in ('T','S'): absolute[field] = weighted_rms(a[field]-ref[field],weights_TS,a['time'],c['duration_s'])
    for field in ('u','v'): absolute[field] = weighted_rms(a[field]-ref[field],a['volume_'+field],a['time'],c['duration_s'])
    amplitude = c['amplitude_m'] or .01
    eta_scale = abs(amplitude*math.sin(c['internal_wave']['q']))
    velocity_scale = amplitude*omega*(c['internal_wave']['q']/c['H_m'])/(math.pi/c['Ly_m'])
    metrics = {'eta_error':absolute['eta']/eta_scale,'u_error':absolute['u']/velocity_scale,'v_error':absolute['v']/velocity_scale,
               'T_error_C':absolute['T'],'S_error_psu':absolute['S']}
    # Independent compensated sum for total inventory at each saved snapshot.
    stocks = {'volume':[math.fsum(float(v) for v in w.reshape(-1)) for w in weights_TS]}
    for field in ('T','S'):
        stocks[field+'_inventory'] = [math.fsum(float(x)*float(w) for x,w in zip(a[field][j].reshape(-1),weights_TS[j].reshape(-1))) for j in range(len(a['time']))]
    for key, values in stocks.items():
        require(values[0] != 0., 'nonzero initial '+key)
        metrics[key] = max(abs(v/values[0]-1.) for v in values)
    thresholds = limits(c)
    failed = [k for k,v in metrics.items() if v>thresholds[k]]
    ratios = {k:metrics[k]/thresholds[k] for k in ('eta_error','u_error','v_error','T_error_C','S_error_psu')}
    initial = {k:float(np.max(np.abs(a[k][0]-ref[k][0]))) for k in ref}
    require(max(initial.values())<5e-11, 'original initial analytic barrier')
    return dict(metrics=metrics,original_limits=thresholds,failed_metrics=failed,normalized_error_ratios=ratios,composite=max(ratios.values()),
                absolute_RMS_before_normalization=absolute,normalizers=dict(eta_m=eta_scale,velocity_m_s=velocity_scale),initial_field_maxima=initial,
                units=dict(T_error_C='degC absolute capacity/time weighted RMS',S_error_psu='psu absolute capacity/time weighted RMS',eta_error='relative to original eta amplitude',u_error='relative to original velocity scale',v_error='relative to original velocity scale',inventory='relative drift'),
                arithmetic='Compensated scalar sums; float64 original arrays. Different rounding from producer retained, not a digit certification.')
