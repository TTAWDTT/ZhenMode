# One-shot pinned filter control; intentionally not a reusable toolkit.
import json, sys, time
from pathlib import Path
import sympy as s
start=time.monotonic()
n=4
R=s.zeros(n-2,n)
for j in range(n-2):
    for k,c in enumerate([1,-2,1]): R[j,j+k]=c
L=s.diag(-3,-2,-2,-3)
for j in range(n-1): L[j,j+1]=L[j+1,j]=1
I=s.eye(n); Ft=I-R.T*R/16; Fn=I-L*L/16
x=s.Matrix([0,1,4,9]); one=s.ones(n,1); alt=s.Matrix([1,-1,1,-1])
y=Ft*x; z=Fn*one
assert y==s.Matrix([-s.Rational(1,8),s.Rational(9,8),s.Rational(33,8),s.Rational(71,8)])
assert z==s.Matrix([s.Rational(5,8),s.Rational(9,8),s.Rational(9,8),s.Rational(5,8)])
wrongt=I-R.T*R/4; wrongn=I-L*L/4
wt=wrongt*alt; wn=wrongn*alt
assert wt==s.Matrix([0,2,-2,0]) and (wt.T*wt)[0]==8 and (alt.T*wrongt*alt)[0]==-4
assert wn==-3*alt and (wn.T*wn)[0]==36
assert Fn*alt==s.zeros(n,1)
assert s.Rational(1,2)+s.Rational(1,3)==s.Rational(5,6)
print('KNOWN_ANSWER rational 1/2+1/3=5/6 PASS')
print('T1 Ft*[0,1,4,9]='+str(list(y))+' PASS negative=-1/8')
print('T2 Fn*ones='+str(list(z))+' PASS nonconstant')
print('FALSE_COPY tangential='+str(list(wt))+' norm2=8 inputnorm2=4 quadratic=-4 CAUGHT')
print('FALSE_COPY normal='+str(list(wn))+' norm2=36 inputnorm2=4 CAUGHT; Fn*Nyquist=0 PASS')
print('CONTROL_PASS elapsed_seconds='+str(time.monotonic()-start)+' sympy='+s.__version__+' python='+sys.version.split()[0])
