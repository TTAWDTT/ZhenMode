def initial(d):return 6+.11*np.exp(.62*d)+.08*d**2+.017*d**4
def exact(d,t,wrong_sign=False):
    foot=2/math.pi*np.arctan(math.exp((1 if wrong_sign else -1)*.21*math.pi*t)*np.tan(math.pi*d/2))
    foot[[0,-1]]=[0.,1.]
    return initial(foot)
def accepted(error,ratios):return error<=.05 and all(r>=1.25 for r in ratios)

def setup(n):
    d=np.linspace(0.,1.,n);h=np.diff(np.r_[0.,.5*(d[:-1]+d[1:]),1.])
    F=np.r_[0.,.21*np.sin(math.pi*.5*(d[:-1]+d[1:])),0.];Nm=-np.diff(F)
    p=SimpleNamespace(dz_node=jnp.asarray(h)[None,None,:],dz_iface=jnp.asarray(np.diff(d))[None,None,:],wet_mask_z=jnp.ones((1,1,n)),dt=DT,adv_nsub=1,vertical_tracer_scheme='muscl_band_v3')
    fj=jnp.asarray(F)[None,None,:];nm=jnp.asarray(Nm)[None,None,:]
    def core(q,reverse=False):
        T=q[None,None,:];flow=-fj if reverse else fj;s=T-T.reshape(-1)[0];baseN=s*nm
        low,boundary=integrated_reference_scalar_rate(T,flow,baseN,nm,p,minimum=0.)
        residual=nm+jnp.diff(flow,axis=-1)
        w=cubic_point_integral(-jnp.diff(flow,axis=-1)/p.dz_node,p)
        w=w.at[...,0].set(flow[...,0]).at[...,-1].set(0.)
        gradient=fourth_uniform_tracer_gradient(s,p)
        target=s[...,0]*flow[...,0]+jnp.sum(p.dz_node*gradient*w,-1)
        magnitude=jnp.sum(jnp.abs(nm),-1,keepdims=True)
        antiN=jnp.abs(nm)/jnp.where(magnitude>0,magnitude,1.)*(target-jnp.sum(baseN,-1))[...,None]
        desired=-w*gradient-T*residual/p.dz_node
        antiZ=point_vertical_antiflux(desired-low,antiN,flow,p)
        delta=-(antiN+jnp.diff(antiZ,axis=-1))/p.dz_node
        alpha,feasible=joint_scalar_blend(T,low,delta,flow,p,minimum=0.)
        rate=low+alpha*delta;horizontal=T*nm+alpha*antiN
        return rate[0,0],horizontal[0,0],boundary[0,0],alpha,feasible,antiZ[0,0]
    return d,h,jax.jit(core),jax.jit(lambda q:core(q,True))
def evolve(n,steps,constant=False):
    d,h,rhs,reverse=setup(n);q=np.ones(n) if constant else initial(d);start=q.copy()
    stock0=np.dot(q.astype(np.longdouble),h.astype(np.longdouble));input_total=np.longdouble(0.)
    maxbudget=0.;constant_rhs=0.;minimum=float(np.min(q));feasible=True;alpha=1.;ticks=[]
    for _ in range(steps):
        tick=time.monotonic();r1,H1,B1,a1,f1,_=map(np.asarray,rhs(jnp.asarray(q)))
        predictor=q+DT*r1;r2,H2,B2,a2,f2,_=map(np.asarray,rhs(jnp.asarray(predictor)))
        euler2=predictor+DT*r2;new=q+DT*(r1+r2)/2
        minimum=min(minimum,float(np.min(predictor)),float(np.min(euler2)));feasible=feasible and bool(f1) and bool(f2);alpha=min(alpha,float(a1),float(a2))
        horizontal=-np.sum(H1.astype(np.longdouble)+H2.astype(np.longdouble))*np.longdouble(DT)/2
        boundary=(np.longdouble(float(B1))+np.longdouble(float(B2)))*np.longdouble(DT)/2
        expected=horizontal+boundary;change=np.dot(new.astype(np.longdouble)-q.astype(np.longdouble),h.astype(np.longdouble))
        maxbudget=max(maxbudget,float(abs(change-expected)));input_total+=expected
        if constant:constant_rhs=max(constant_rhs,float(np.max(abs(r1))),float(np.max(abs(r2))))
        q=new;ticks.append(time.monotonic()-tick)
    stock=np.dot(q.astype(np.longdouble),h.astype(np.longdouble));negative=float(np.max(abs(np.asarray(reverse(jnp.ones(n))[0]))))
    budget=dict(stock_initial=float(stock0),stock_final=float(stock),horizontal_plus_boundary=float(input_total),inventory_residual=float(stock-stock0-input_total),budget_limit=1e-12*max(1.,abs(float(stock0)),abs(float(input_total))),maximum_step_budget=maxbudget,constant_rhs_max=constant_rhs,reversed_mass_constant_defect=negative,minimum_Euler=minimum,all_feasible=feasible,minimum_alpha=alpha)
    return d,h,start,q,budget,ticks
