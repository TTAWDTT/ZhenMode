def utc(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
def sha(data): return hashlib.sha256(data).hexdigest()

def require(c, msg):

def npz(path):
    with zipfile.ZipFile(path) as z: require(sum(i.file_size for i in z.infolist()) <= 128*1024**2, 'NPZ decoded budget')
    with np.load(path, allow_pickle=False) as z: a = {k: z[k] for k in z.files}
    if 'metadata' in a: a['metadata'] = json.loads(str(a['metadata']))
    return a
def same(a,b): return a.shape == b.shape and a.dtype == b.dtype and np.array_equal(a,b)
def close(a,b, inventory=False): return abs(a-b) <= (2e-14 if inventory else 5e-15 + 5e-12*abs(b))
def rms(error, weight, times, duration):
    # Independently accumulate a spatial mean square at each snapshot, then time trapezoids.
    means=[]
    for j in range(len(times)):
        w=np.broadcast_to(weight[j],error[j].shape).ravel()
        e=error[j].ravel()
        means.append(float(np.dot(w,e*e)/np.sum(w)))
    return math.sqrt(sum((means[j-1]+means[j])*(times[j]-times[j-1])/2 for j in range(1,len(times)))/duration)

def controls():

        require(read(HERE/'controls-result.json')['pass_all'], 'preserved controls failed')

    t=np.array([0.,1.,2.]); e=np.broadcast_to(t[:,None,None],(3,2,2)).copy(); w=np.ones_like(e)
    stream=io.BytesIO(); np.savez_compressed(stream,time=t,error=e,weight=w); stream.seek(0)
    with np.load(stream,allow_pickle=False) as z: value=rms(z['error'],z['weight'],z['time'],2.)

def reference(c,a):
    # Direct pointwise form for ocean-solver nodal output; original linear reference.
    gamma=c['internal_wave']['dT_dz_C_m'];q=c['internal_wave']['q'];m=q/c['H_m'];k=math.pi/c['Ly_m'];A=c['amplitude_m']
    om=math.hypot(c['f'],math.sqrt(c['internal_wave']['N2_s_2'])*k/m)
    z=-np.arange(c['nz'])*c['H_m']/(c['nz']-1)
    cost=np.cos(om*a['time']);sint=np.sin(om*a['time'])
    Tbase=(c['T_C']+gamma*z)[None,None,:]
    Tdelta=-gamma*A*np.sin(m*(z+c['H_m']))[None,None,:]*np.cos(k*a['y_eta'])[None,:,None]*cost[:,None,None]
    ref={'T':Tbase+Tdelta,'S':np.full_like(a['S'],c['S_psu']), 'eta':A*math.sin(q)*cost[:,None]*np.cos(k*a['y_eta'])[None,:]}
    for v in ('u','v'):
        spatial=np.sin(k*a['y_'+v])[:,None]*np.cos(m*(z+c['H_m']))[None,:]
        ref[v]=A*m/k*spatial[None,:,:]*((-c['f']*cost) if v=='u' else om*sint)[:,None,None]
    return ref,om

def metric_readback(c,a):
    if c['benchmark']=='thermal-wind': return thermal_readback(c,a)
    ref,omega=reference(c,a);t=a['time']; vol=a['h']*a['area'][None,:,None]
    metrics={}
    mode=c['internal_wave'];A=c['amplitude_m'] or .01;m=mode['q']/c['H_m'];k=math.pi/c['Ly_m']
    metrics['eta_error']=rms(a['eta']-ref['eta'],np.broadcast_to(a['area'],a['eta'].shape),t,c['duration_s'])/abs(A*math.sin(mode['q']))
    for v in ('u','v'): metrics[v+'_error']=rms(a[v]-ref[v],a['volume_'+v],t,c['duration_s'])/(A*omega*m/k)
    for v,key in [('T','T_error_C'),('S','S_error_psu')]:metrics[key]=rms(a[v]-ref[v],vol,t,c['duration_s'])
    sums=vol.sum(axis=(1,2));metrics['volume']=float(np.max(np.abs(sums/sums[0]-1)))
    for v in ('T','S'):
        totals=(a[v]*vol).sum(axis=(1,2));metrics[v+'_inventory']=float(np.max(np.abs(totals/totals[0]-1)))
    limits={v+'_error':c['thresholds']['error'] for v in ('eta','u','v')}
    limits.update({v:c['thresholds'][v] for v in ('T_error_C','S_error_psu','volume')})
    limits.update({v+'_inventory':c['thresholds']['tracer_inventory'] for v in ('T','S')})
    failed=[k for k,v in metrics.items() if v>limits[k]]
    null={}
    if c['control']=='null':
        for v in ('eta','u','v','T','S','h'):
            null[v]=float(np.max(np.abs(a[v]-a[v][:1])))
            if null[v]>c['thresholds']['null_absolute']:failed.append('null_'+v)
    ratios={v:metrics[v]/limits[v] for v in ('eta_error','u_error','v_error','T_error_C','S_error_psu')}
    initial={v:float(np.max(np.abs(a[v][0]-ref[v][0]))) for v in ref}
    require(max(initial.values())<5e-11,'initial analytic fields')
    return dict(metrics=metrics,limits=limits,failed_metrics=failed,normalized_error_ratios=ratios,composite=max(ratios.values()),null_absolute_errors=null,initial_errors=initial)
