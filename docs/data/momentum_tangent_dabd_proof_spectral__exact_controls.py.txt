#!/usr/bin/env python3
"""One-shot pin-specific author controls, not an independent certifier."""
import os, resource, signal, threading, time, json, pathlib
START = time.monotonic()
os.sched_setaffinity(0, {min(os.sched_getaffinity(0))})
resource.setrlimit(resource.RLIMIT_CPU, (180, 180))
resource.setrlimit(resource.RLIMIT_AS, (4 * 1024**3, 4 * 1024**3))
signal.alarm(180)
rss_samples = []
done = threading.Event()
def sample():
    while not done.is_set():
        for line in pathlib.Path('/proc/self/status').read_text().splitlines():
            if line.startswith('VmRSS:'):
                rss_samples.append(int(line.split()[1]) * 1024)
        done.wait(0.02)
monitor = threading.Thread(target=sample, daemon=True)
monitor.start()
import sympy as s
out = pathlib.Path(__file__).parent
checks = []
def check(name, predicate, value=None):
    assert predicate, name
    checks.append({'name':name, 'PASS':True, 'value':str(value) if value is not None else None})
    (out/'controls.json').write_text(json.dumps(checks,indent=2))
Q=s.Rational
check('known-answer-rational', Q(1,2)+Q(1,3)==Q(5,6))
n=4
R=s.zeros(2,4)
for j in range(2):
    R[j,j], R[j,j+1], R[j,j+2] = 1,-2,1
L=s.diag(-3,-2,-2,-3)
for j in range(3): L[j,j+1]=L[j+1,j]=1
Ft=s.eye(n)-R.T*R/16
Fn=s.eye(n)-L*L/16
B=(s.eye(n)+L/4)**2
one=s.ones(4,1); affine=s.Matrix(range(4)); alt=s.Matrix([1,-1,1,-1])
x1=s.Matrix([0,1,4,9]); t1=s.Matrix([Q(-1,8),Q(9,8),Q(33,8),Q(71,8)])
t2=s.Matrix([Q(5,8),Q(9,8),Q(9,8),Q(5,8)])
check('T1', Ft*x1==t1, list(Ft*x1))
check('T2', Fn*one==t2, list(Fn*one))
check('comparator-mismatch-caught', Ft*x1!=t1+s.Matrix([1,0,0,0]))
def contract_psd(F):
    return F==F.T and F.is_positive_semidefinite is True and (s.eye(4)-F).is_positive_semidefinite is True and (s.eye(4)-F.T*F).is_positive_semidefinite is True
check('clean-Ft-detector',contract_psd(Ft))
check('clean-Fn-detector',contract_psd(Fn))
wrongt=s.eye(4)-R.T*R/4; wrongn=s.eye(4)-L*L/4
check('tangential-plant-output',wrongt*alt==s.Matrix([0,2,-2,0]),list(wrongt*alt))
check('tangential-plant-norm2', (wrongt*alt).dot(wrongt*alt)==8 and alt.dot(alt)==4)
check('tangential-plant-quadratic', alt.dot(wrongt*alt)==-4)
check('tangential-plant-CAUGHT',not contract_psd(wrongt))
check('normal-plant-output',wrongn*alt==-3*alt,list(wrongn*alt))
check('normal-plant-norm2',(wrongn*alt).dot(wrongn*alt)==36)
check('normal-plant-CAUGHT',not contract_psd(wrongn))
check('zero-null',Ft*s.zeros(4,1)==s.zeros(4,1) and Fn*s.zeros(4,1)==s.zeros(4,1))
check('Ft-affine-and-sum', R*one==s.zeros(2,1) and R*affine==s.zeros(2,1) and Ft*one==one and Ft*affine==affine and one.T*Ft==one.T)
check('Ft-kernel-dimension',len(R.nullspace())==2)
v=s.Matrix(s.symbols('v0:4')); d=[v[j]-2*v[j+1]+v[j+2] for j in range(2)]
res=list(v)
for j,r in enumerate(d):
    res[j]-=r/16; res[j+1]+=r/8; res[j+2]-=r/16
check('Ft-loop-is-matrix',s.Matrix(res)==Ft*v)
def Lapply(w):
    return s.Matrix([-3*w[0]+w[1],w[0]-2*w[1]+w[2],w[1]-2*w[2]+w[3],w[2]-3*w[3]])
check('Fn-loop-is-matrix', s.expand(Lapply(v))==L*v and s.expand(v-Lapply(Lapply(v))/16)==Fn*v)
S=s.Matrix(4,4,lambda j,k:s.sin((k+1)*s.pi*(s.Rational(2*j+1,2))/4))
lam=[-4*s.sin(k*s.pi/8)**2 for k in range(1,5)]
check('n4-sine-residual', (L*S-S*s.diag(*lam)).applyfunc(s.simplify)==s.zeros(4))
check('n4-sine-orthogonal', (S.T*S).applyfunc(s.simplify)==s.diag(2,2,2,4))
check('n4-filter-spectral-map',(Fn*S-S*s.diag(*[1-v*v/16 for v in lam])).applyfunc(s.simplify)==s.zeros(4))
check('n4-B-spectral-map',(B*S-S*s.diag(*[(1+v/4)**2 for v in lam])).applyfunc(s.simplify)==s.zeros(4))
check('Nyquist-killed',L*alt==-4*alt and Fn*alt==s.zeros(4,1))
theta,u=s.symbols('theta u',real=True)
rec=s.sin(u-theta)-2*s.sin(u)+s.sin(u+theta)+4*s.sin(theta/2)**2*s.sin(u)
check('general-sine-recurrence',s.trigsimp(s.expand_trig(rec))==0)
check('left-odd-ghost',s.sin(-theta/2)==-s.sin(theta/2))
kk=s.symbols('k',integer=True)
check('right-odd-ghost',s.expand_trig(s.sin(kk*s.pi+theta/2)+s.sin(kk*s.pi-theta/2))==0)
z=s.symbols('z',nonzero=True)
mult=s.expand((1-2*z+z*z)*(1-2/z+1/z**2))
check('convolution-Laurent-symbol',s.expand(mult-(2-z-1/z)**2)==0)
check('normal-F-symbol',1-(-4*s.sin(theta/2)**2)**2/16==1-s.sin(theta/2)**4)
check('B-symbol',s.trigsimp((1-s.sin(theta/2)**2)**2-s.cos(theta/2)**4)==0)
fdeparture=s.series(s.sin(theta/2)**4,theta,0,6)
bdeparture=s.series(1-s.cos(theta/2)**4,theta,0,4)
check('low-wave-departures',fdeparture.removeO()==theta**4/16 and bdeparture.removeO()==theta**2/2, [fdeparture,bdeparture])
TV=lambda w:sum(abs(w[j+1]-w[j]) for j in range(3))
check('Ft-not-positivity',min(t1)<0)
e0=s.Matrix([1,0,0,0]); fe0=s.Matrix([Q(15,16),Q(1,8),Q(-1,16),0])
check('Ft-TVD-witness-output',Ft*e0==fe0,list(fe0))
check('Ft-not-TVD',TV(fe0)>TV(e0),[TV(e0),TV(fe0)])
check('Fn-not-positivity',Fn[0,2]<0,Fn[0,2])
check('Fn-not-TVD',TV(t2)>TV(one),[TV(one),TV(t2)])
check('quadratic-Gram-identity',s.expand((v.T*R.T*R*v)[0]-(R*v).dot(R*v))==0)
a=s.symbols('a',real=True)
check('B-difference-symbol',s.expand((1-a**4)-(1-a**2)**2-2*a**2*(1-a**2))==0)
check('n4-normal-B-Loewner-comparison',(Fn-B).is_positive_semidefinite is True)
done.set(); monitor.join()
receipt={'grade':'AUTHOR-CONTROLS-PASS','n_values':[4],'checks':len(checks),'denominator':38,'sympy':s.__version__,'wall_seconds':time.monotonic()-START,'cpu_seconds':time.process_time(),'affinity':list(os.sched_getaffinity(0)),'cpu_limit_seconds':180,'wall_limit_seconds':180,'as_limit_bytes':4*1024**3,'sampled_peak_rss_bytes':max(rss_samples),'rss_samples':len(rss_samples),'maxrss_bytes':resource.getrusage(resource.RUSAGE_SELF).ru_maxrss*1024}
assert len(checks)==receipt['denominator'],len(checks)
(out/'receipt.json').write_text(json.dumps(receipt,indent=2))
print('AUTHOR-CONTROLS-PASS n=4 checks=%s/%s'%(len(checks),receipt['denominator']))
print('T1='+str(list(t1))+'; T2='+str(list(t2)))
print('PLANTS CAUGHT: Ft norm2=8>4, quadratic=-4; Fn norm2=36>4')
print('TIMING wall_seconds=%.6f cpu_seconds=%.6f sampled_peak_rss_bytes=%s samples=%s'%(receipt['wall_seconds'],receipt['cpu_seconds'],receipt['sampled_peak_rss_bytes'],receipt['rss_samples']))
