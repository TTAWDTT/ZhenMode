def relative(a,b):
    den=np.linalg.norm(b.ravel())
    return float(np.linalg.norm((a-b).ravel())/den) if den else float(np.linalg.norm((a-b).ravel()))
def augment(operator,h0,hc,hs,nu):
    d=operator.shape[0]; aug=np.zeros((d+3,d+3))
    aug[:d,:d]=operator
    aug[:d,d]=h0;aug[:d,d+1]=hc;aug[:d,d+2]=hs
    aug[d+1,d+2]=-nu;aug[d+2,d+1]=nu
    initial=np.zeros(d+3);initial[d:d+3]=[1,1,0]
    return aug,initial
def solve(operator,h0,hc,hs,nu,times):
    aug,initial=augment(operator,h0,hc,hs,nu)
    return np.stack([(expm(aug*float(t))@initial)[:len(h0)] for t in times])

k=np.pi/c['Ly_m'];m=q/HH;omega=np.sqrt(ff*ff+NN*(k/m)**2);K=2*k;nu=2*omega;alpha=NN/(gg*gam)

operator[:n,n:2*n]=ff*eye;operator[n:2*n,:n]=-ff*eye
operator[n:2*n,2*n:3*n]=-K*Qt
operator[n,3*n]=K*gg
operator[2*n:3*n,n:2*n]=NN*K*Qb
operator[3*n,n]=-K*HH
weights=np.concatenate([M,M,M/NN,[gg/HH]]);D=np.sqrt(weights)
