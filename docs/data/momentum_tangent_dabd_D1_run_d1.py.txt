        depth=wet.copy(),wet_mask=wet,ocean_mask=ocean,land_mask=~ocean,

    zero=('nu_h','nu_v','nu_bi','kappa_h','kappa_v','kappa_bi','kappa_conv','kappa_gm','kappa_redi','r_bot','cd','tau_x','tau_y','Q_heat')

    effective={}
    for name in p._fields:

        if value is None or isinstance(value,(str,bool,int,float)):
            effective[name]=value
        else:
            a=np.asarray(value)
            effective[name]=dict(shape=list(a.shape),dtype=str(a.dtype),sha256=hashlib.sha256(a.tobytes()).hexdigest(),
                                 min=float(a.min()) if a.size else None,max=float(a.max()) if a.size else None)
    return p,z,effective


def fields(n,phase,f=.1,A=1e-4):
    y=(np.arange(n)+.5)/n;z=-np.arange(9)/8;Z=z+1
    sine=np.sin(math.pi*y)[:,None];cosine=np.cos(math.pi*y)[:,None]
    cz=np.cos(math.pi*Z)[None,:];sz=np.sin(math.pi*Z)[None,:]
    u=-A*f*cz*sine*math.cos(phase)
    v=A*cz*sine*math.sin(phase)
    w=-A*sz*cosine*math.sin(phase)
    return tuple(jnp.asarray(np.broadcast_to(q[None,:,:],(8,n,9)).copy()) for q in (u,v,w))


def evaluate(n,phase,p):
    u,v,wa=fields(n,phase)
    zeros=jnp.zeros_like(u)
    state=State(u,v,zeros,zeros,jnp.zeros((8,n)),jnp.zeros((8,n)))
    wn=transport._compute_vertical_velocity(state,p)
    gu=horizontal._gradient_face_gated_3d(u,p)[1]
    gv=horizontal._gradient_face_gated_3d(v,p,normal=True)[1]
    gzu=transport._d_dz(transport._fill_ghost_bottom(u,p),p)
    gzv=transport._d_dz(transport._fill_ghost_bottom(v,p),p)
    aa=transport._advection_flux_form(u,v,wa,p)
    an=transport._advection_flux_form(u,v,wn,p)
    result={'u':u,'v':v,'w_analytic':wa,'w_native':wn,'grad_u':gu,'grad_v':gv,'grad_z_u':gzu,'grad_z_v':gzv,
            'adv_analytic_u':aa[0],'adv_analytic_v':aa[1],'adv_native_u':an[0],'adv_native_v':an[1]}
    return {key:np.asarray(value) for key,value in result.items()}


class Gate:
    def __init__(self,out):
        self.out=out;self.rows=[];self.groups=set();self.mutations=set()
    def check(self,name,actual,expected,scale=1.,group=None):
        a,b=np.asarray(actual),np.asarray(expected)
        scale=max(1.,float(scale));bound=128*np.finfo(np.float64).eps*scale
        residual=float(np.max(np.abs(a-b)))
        ok=bool(a.shape==b.shape and np.all(np.isfinite(a)) and np.all(np.isfinite(b)) and residual<=bound)
        row=dict(name=name,shape=list(a.shape),max_abs=residual,sum_abs_terms=scale,budget=bound,pass_gate=ok)
        self.rows.append(row)
        with (self.out/'gates.jsonl').open('a') as f:f.write(json.dumps(row)+'\n')
        if not ok:raise ArithmeticError('FAIL '+name)
        if group and group not in self.mutations:
            bad=b.copy();bad.flat[0]+=1.
            rejected=not(np.max(np.abs(a-bad))<=bound)
            self.rows.append(dict(name=name+'__planted_plus_one',group=group,caught=bool(rejected),expected_failure=True))
            with (self.out/'gates.jsonl').open('a') as f:f.write(json.dumps(self.rows[-1])+'\n')
            if not rejected:raise ArithmeticError('uncaught mutation '+group)
            self.mutations.add(group)
        if group:self.groups.add(group)


def controls(g,p,n,z,truth):

    h=1/n;s=np.sin(math.pi*(np.arange(n)+.5)/n);c=np.cos(math.pi*(np.arange(n)+.5)/n)
    q=np.sin(2*math.pi*(np.arange(n)+.5)/n)

    one=np.ones(n);linear=(np.arange(n)+.5)*h

    g.check('sine_delta_south',np.array([de[0]-do[0]]),np.array([-s[0]/h]),scale=2/h,group='gradient_identity')
    be,bo=bf(q,False),bf(q,True)

    g.check('filter_delta_south',be[:2]-bo[:2],np.array([(9*q[0]+q[1])/16,q[0]/8]),scale=sum(abs(q)),group='filter_delta')
    g.check('filter_delta_north',be[-2:]-bo[-2:],np.array([q[-1]/8,(9*q[-1]+q[-2])/16]),scale=sum(abs(q)),group='filter_delta_north')
    ce=bf(c,False);cv=np.asarray(o.binomial(c.tolist(),'even'))

    dv=np.asarray(channel.channel_divergence(lift(np.zeros(n)),lift(s),p))[0,:,0]

    g.check('continuity_sin_symbol',dv,k4*c,scale=12/h,group='continuity')
    g.check('pressure_independent_index',gp,np.array(o.gradient4(c.tolist(),h)),scale=12/h,group='pressure_index')
    g.check('continuity_independent_index',dv,np.array(o.divergence4(s.tolist(),h)),scale=12/h,group='continuity_index')
    dispatched=np.asarray(horizontal._gradient_conservative_3d(lift(c),p)[1])[0,:,0]
    g.check('pressure_dispatch',dispatched,gp,scale=12/h,group='pressure_dispatch')
    faces=transport._layer_face_transports(lift(np.zeros(n)),lift(s),p)
    transport_div=np.asarray(transport._face_transport_divergence(*faces,p)/p.dz_node)[0,:,0]

    g.check('closed_north_face',np.asarray(faces[1])[:,-1,:],np.zeros((8,9)),group='north_face')
    g.check('closed_south_odd_ghost',np.asarray(horizontal._mirror_latitude(lift(s),normal=True))[:,0,:]
            +np.asarray(lift(s))[:,0,:],np.zeros((8,9)),group='south_face')
    work=float(np.dot(c,dv)+np.dot(s,gp))

    mp_rows={item['ny']:item for item in truth['precisions']['160']}[n]

def region_stats(array):
    a=np.asarray(array)[0];n=a.shape[0]
    rows={**{f'south_{i}':slice(i,i+1) for i in range(3)},
          **{f'north_{i}':slice(n-1-i,n-i) for i in range(3)},'interior':slice(3,n-3)}
    levels={'top':slice(0,1),'interior':slice(1,8),'bottom':slice(8,9)}
    return {r+'__'+z:dict(max_abs=float(np.max(abs(a[j,k]))),signed_mean=float(np.mean(a[j,k])))
            for r,j in rows.items() for z,k in levels.items()}


def primary(g,p,n,phase,z,index):
    import stencil_oracle as o
    data=evaluate(n,phase,p)
    for role in ('analytic','native'):
        rawu,rawv,gyu,gyv,gzu,gzv=o.raw_advection(data['u'][0].tolist(),data['v'][0].tolist(),
            data['w_'+role][0].tolist(),z.tolist(),1/n)
        eu=np.asarray(o.filter_matrix(rawu,'edge'));ev=np.asarray(o.filter_matrix(rawv,'odd'))
        scale=o.term_scale(rawu)+o.term_scale(rawv)
        for component,gy,gz in [('u',data['grad_u'],data['grad_z_u']),('v',data['grad_v'],data['grad_z_v'])]:
            data['horizontal_'+role+'_'+component]=-data['v']*gy
            data['vertical_'+role+'_'+component]=-data['w_'+role]*gz
            data['raw_native_helpers_'+role+'_'+component]=data['horizontal_'+role+'_'+component]+data['vertical_'+role+'_'+component]
        g.check(role+'_raw_u',data['raw_native_helpers_'+role+'_u'][0],np.asarray(rawu),scale=scale,group='raw_u_'+role)
        g.check(role+'_raw_v',data['raw_native_helpers_'+role+'_v'][0],np.asarray(rawv),scale=scale,group='raw_v_'+role)
        g.check(role+'_full_u',data['adv_'+role+'_u'][0],eu,scale=scale,group='full_u_'+role)
        g.check(role+'_full_v',data['adv_'+role+'_v'][0],ev,scale=scale,group='full_v_'+role)
        data['raw_'+role+'_u']=np.broadcast_to(np.asarray(rawu)[None],(8,n,9))
        data['raw_'+role+'_v']=np.broadcast_to(np.asarray(rawv)[None],(8,n,9))
        data['expected_'+role+'_u']=np.broadcast_to(eu[None],(8,n,9))
        data['expected_'+role+'_v']=np.broadcast_to(ev[None],(8,n,9))
        data['residual_'+role+'_u']=data['adv_'+role+'_u']-data['expected_'+role+'_u']
        data['residual_'+role+'_v']=data['adv_'+role+'_v']-data['expected_'+role+'_v']
        if role=='analytic':
            g.check('primary_gradient_u',data['grad_u'][0],np.asarray(gyu).T,scale=1.,group='primary_gu')
            g.check('primary_gradient_v',data['grad_v'][0],np.asarray(gyv).T,scale=1.,group='primary_gv')
            g.check('primary_gradient_z_u',data['grad_z_u'][0],np.asarray(gzu),scale=1.,group='primary_gzu')
            g.check('primary_gradient_z_v',data['grad_z_v'][0],np.asarray(gzv),scale=1.,group='primary_gzv')
            raw_odd=o.raw_advection(data['u'][0].tolist(),data['v'][0].tolist(),data['w_analytic'][0].tolist(),z.tolist(),1/n,'odd')[0]
            expected_delta=np.zeros((n,9));expected_delta[0]=data['v'][0,0]*data['u'][0,0]*n
            expected_delta[-1]=-data['v'][0,-1]*data['u'][0,-1]*n
            g.check('primary_raw_wall_delta',np.asarray(rawu)-np.asarray(raw_odd),expected_delta,scale=scale,group='raw_wall_delta')
            data['predicted_raw_wall_delta_u']=np.broadcast_to(expected_delta[None],(8,n,9))
            data['counterfactual_odd_gradient_raw_u']=np.broadcast_to(np.asarray(raw_odd)[None],(8,n,9))
            data['counterfactual_odd_gradient_and_filter_u']=np.broadcast_to(np.asarray(o.filter_matrix(raw_odd,'odd'))[None],(8,n,9))
    y=(np.arange(n)+.5)/n
    exactu=1e-8*.1*math.pi/4*np.sin(2*math.pi*y)*math.sin(2*phase)
    exactv=-1e-8*math.pi/4*np.sin(2*math.pi*y)*(1-math.cos(2*phase))
    for component,exact in [('u',exactu),('v',exactv)]:
        field=np.broadcast_to(exact[None,:,None],(8,n,9)).copy()
        data['continuous_'+component]=field
        data['continuous_error_analytic_'+component]=data['adv_analytic_'+component]-field
        for normal,label in [(False,'edge'),(True,'odd')]:
            data['continuous_product_filter_'+component+'_'+label]=np.asarray(horizontal._dealias_h_fd(jnp.asarray(field),p,normal=normal))
    data['w_native_minus_analytic']=data['w_native']-data['w_analytic']
    stem=f'case{index:02d}_ny{n}_phase{phase:.16g}'
    np.savez(g.out/(stem+'.npz'),**data)
    stats={name:region_stats(value) for name,value in data.items() if ('error' in name or 'residual' in name or name=='w_native_minus_analytic')}
    (g.out/(stem+'.json')).write_text(json.dumps(dict(ny=n,phase=phase,kind='DEVELOPMENT descriptive continuous error; no continuous PASS gate',statistics=stats),indent=2)+'\n')
    print(json.dumps({'stage':'primary','case':index,'ny':n,'phase':phase,'artifact':stem}),flush=True)


def main():

    init_native();native_ready=time.monotonic();modules_before=modules()
    (out/'modules-before.json').write_text(json.dumps(modules_before,indent=2)+'\n')
    try:
        if args.mode=='timing':
            p,z,effective=params(8)
            init_done=time.monotonic();evaluate(8,math.pi/6,p);first=time.monotonic()
            print(json.dumps({'stage':'first_helper','elapsed_s':first-start}),flush=True)
            evaluate(8,math.pi/6,p);second=time.monotonic()
            timing=dict(total_s=second-start,identity_and_import_s=native_ready-start,initialization_s=init_done-native_ready,
                        first_case_s=first-init_done,warm_case_s=second-first,continuous_oracle_read=False,comparison_performed=False)
            (out/'timing.json').write_text(json.dumps(timing,indent=2)+'\n');print(json.dumps(timing),flush=True)
        else:
            truth=json.loads((ROOT/'truth'/'truth.json').read_text());g=Gate(out);count=0
            for n in (8,16):
                p,z,effective=params(n);(out/f'effective-ny{n}.json').write_text(json.dumps(effective,indent=2)+'\n')
                controls(g,p,n,z,truth)
                print(json.dumps({'stage':'controls','ny':n,'positive_rows':sum('pass_gate' in r for r in g.rows),'mutations':len(g.mutations)}),flush=True)
                for phase in (math.pi/6,math.pi/3):
                    count+=1;primary(g,p,n,phase,z,count)
            assert count==4 and g.groups==g.mutations
            summary=dict(verdict='FINITE STENCIL DEVELOPMENT PASS; physical attribution OPEN',primary_count=count,
                expected_primary_count=4,positive_gate_count=sum('pass_gate' in r for r in g.rows),mutation_groups=sorted(g.mutations),
                continuous_errors_graded=False,elapsed_s=time.monotonic()-start)
            (out/'summary.json').write_text(json.dumps(summary,indent=2)+'\n');print(json.dumps(summary),flush=True)
    finally:
        after=identity();assert before==after
        (out/'source-after.json').write_text(json.dumps(after,indent=2)+'\n')
        module_after=modules();(out/'modules-after.json').write_text(json.dumps(module_after,indent=2)+'\n')
        assert all(module_after[k]==v for k,v in modules_before.items())
